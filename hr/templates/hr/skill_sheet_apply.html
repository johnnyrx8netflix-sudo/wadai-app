async function handlePostalInput() { const postalInput = document.getElementById("id_postal_code"); const addressInput = document.getElementById("id_address"); const pHint = document.getElementById("postal-hint"); const addrHint = document.getElementById("address-hint"); if (!postalInput || !addressInput) return; const formatted = formatPostalHyphen(postalInput.value); if (postalInput.value !== formatted) { postalInput.value = formatted; } const zip = formatted.replace(/\D/g, ""); if (zip.length < 7) { if (pHint) { delete pHint.dataset.autoState; const lang = localStorage.getItem("skillsheet_lang") || "ja"; pHint.textContent = (uiTexts[lang] || uiTexts.ja).postalHint; pHint.className = "hint"; } return; } const lang = localStorage.getItem("skillsheet_lang") || "ja"; const t = uiTexts[lang] || uiTexts.ja; let fullAddress = ""; let networkFailed = false; try { const res = await fetch(`/api/postal-lookup/?zipcode=${encodeURIComponent(zip)}`); if (res.ok) { const data = await res.json(); if (data && data.ok && data.address) { fullAddress = data.address; } } else { networkFailed = true; } } catch (e) { networkFailed = true; } if (!fullAddress) { try { const res2 = await fetch(`https://zipcloud.ibsnet.co.jp/api/api/json?zipcode=${zip}`); if (res2.ok) { const data2 = await res2.json(); if (data2.results && data2.results.length > 0) { const r = data2.results[0]; fullAddress = `${r.address1 || ""}${r.address2 || ""}${r.address3 || ""}`; } networkFailed = false; } } catch (e2) {} } if (fullAddress) { addressInput.value = fullAddress; addressInput.classList.add("auto-filled"); addressInput.dispatchEvent(new Event("input", { bubbles: true })); addressInput.dispatchEvent(new Event("change", { bubbles: true })); if (pHint) { pHint.dataset.autoState = "ok"; pHint.textContent = `${t.postalFound}: ${fullAddress}`; pHint.className = "hint status-ok"; } if (addrHint) { addrHint.dataset.autoState = "ok"; addrHint.textContent = t.addressHint; addrHint.className = "hint status-ok"; } } else if (pHint) { pHint.dataset.autoState = "err"; pHint.textContent = networkFailed ? t.postalErr : t.postalNotFound; pHint.className = "hint status-err"; } }
